instCons :: VirtualMachine () Source #

Implements CONS (Opcode 0x90).

@details Constructs a new list by prepending an element. Stack: [..., elem, list] -> [..., new_list] Note: Pops list first (Right), then elem (Left).

instHead :: VirtualMachine () Source #

Implements HEAD (Opcode 0x91).

@details Extracts the first element of a list. Stack: [..., list] -> [..., head]

@throws Error if list is empty.

instTail :: VirtualMachine () Source #

Implements TAIL (Opcode 0x92).

@details Extracts the rest of the list (minus the first element). Stack: [..., list] -> [..., tail]

@throws Error if list is empty.
